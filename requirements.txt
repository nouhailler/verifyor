# Verifyor — Dépendances Node.js
# Projet : plateforme de vérification et d'intelligence email
# Runtime : Node.js >= 18 (LTS recommandé)
#
# Ce fichier documente toutes les dépendances du projet.
# Les dépendances npm sont gérées via package.json / package-lock.json.
# Installer avec : npm install

# ─────────────────────────────────────────────
#  DÉPENDANCES NPM (production)
# ─────────────────────────────────────────────

express@^4.21.2          # Framework HTTP — routage, middleware, service de fichiers statiques
dotenv@^16.4.5           # Chargement des variables d'environnement depuis .env

# ─────────────────────────────────────────────
#  MODULES NATIFS NODE.JS (inclus dans Node >= 18)
# ─────────────────────────────────────────────

node:dns/promises        # Résolution DNS asynchrone — MX, TXT, NS, SPF, DMARC
node:net                 # Sockets TCP — SMTP handshake avancé
node:crypto              # Hash SHA-256 — lookup Gravatar
node:fs                  # Lecture/écriture fichiers — .env, base SQLite
node:path                # Résolution de chemins cross-platform
node:sqlite              # Base de données SQLite embarquée (expérimental Node 22+)

# ─────────────────────────────────────────────
#  SERVICES EXTERNES (clés API optionnelles)
# ─────────────────────────────────────────────

# ZeroBounce API   https://www.zerobounce.net/
#   Variable env : ZEROBOUNCE_API_KEY
#   Usage         : vérification email cloud (provider "zerobounce")

# Abstract API     https://www.abstractapi.com/api/email-verification
#   Variable env : ABSTRACT_API_KEY
#   Usage         : vérification email cloud alternative (provider "abstract")

# Hunter.io API    https://hunter.io/api
#   Variable env : HUNTER_API_KEY
#   Usage         : enrichissement B2B — recherche domaine, pattern email, données entreprise

# Gravatar API     https://gravatar.com/
#   Variable env : GRAVATAR_API_KEY (optionnel — lookup public sans clé possible)
#   Usage         : enrichissement social — avatar et profil public associés à l'email

# ─────────────────────────────────────────────
#  RÉSUMÉ DES VERSIONS MINIMALES
# ─────────────────────────────────────────────

# Node.js  >= 18.0.0   (LTS — pour node:sqlite >= 22.5.0 requis)
# npm      >= 8.0.0
# dpkg     >= 1.18      (pour construire le paquet Debian uniquement)
